# Differences between the baseline and candidate builds that phase A is
# meant to cause. lib/artifact_diff.py reports every other difference as
# UNEXPECTED, and lists the rules below that matched nothing.
#
# Format: <kind> <regex>  # reason
# The regex must match the whole subject:
#   dist-file-*          path in the distribution
#   jar-entry-*          <jar path in the distribution>!<entry>
#   class-*              <command>:<class name>   (command = hadoop, hdfs, yarn,
#                        mapred, tools, yarn-timelinereader, client,
#                        client-minicluster)
#   dep-*                <module path>|<group:artifact:type[:classifier]:version:scope>
#
# Keep the rules narrow: a broad pattern can hide a real regression.

# --- HADOOP-20013: the duplicate javax.annotation-api 1.3.2 is dropped.
# jakarta.annotation-api 1.3.5, which provides the same javax.annotation
# classes, was already in the distribution and every module that loses the
# javax copy already resolved it: no jar is added, no dependency is added and
# no class changes the jar it is loaded from (none of these show up in CI).
dist-file-removed share/hadoop/[a-z]+/lib/javax\.annotation-api-1\.3\.2\.jar  # HADOOP-20013: duplicate javax.annotation jar dropped
dep-removed [^|]+\|javax\.annotation:javax\.annotation-api:jar:1\.3\.2:(compile|runtime|provided|test)  # HADOOP-20013

# --- License files list the changed third-party artifacts.
dist-file-changed (LICENSE-binary|NOTICE-binary)  # HADOOP-20013/20014: license entries for the swapped artifacts

# --- HADOOP-20016 / HADOOP-20015: jetty-http and jetty-webapp declared where
# they are imported. They were already resolved there transitively with the
# same version and scope, so no difference is expected (none shows up in CI).

# --- HADOOP-20014: Jersey test container moves from Jetty to the JDK HTTP
# server. Test scope only; the unit-test phase covers the effect.
dep-removed [^|]+\|org\.glassfish\.jersey\.test-framework\.providers:jersey-test-framework-provider-jetty:jar:[^:]+:test  # HADOOP-20014
dep-removed [^|]+\|org\.glassfish\.jersey\.containers:jersey-container-jetty-http:jar:[^:]+:test  # HADOOP-20014: came with the Jetty test provider
dep-added [^|]+\|org\.glassfish\.jersey\.test-framework\.providers:jersey-test-framework-provider-jdk-http:jar:[^:]+:test  # HADOOP-20014
dep-added [^|]+\|org\.glassfish\.jersey\.containers:jersey-container-jdk-http:jar:[^:]+:test  # HADOOP-20014: comes with the JDK HTTP test provider
dep-removed [^|]+\|org\.eclipse\.jetty:jetty-[a-z\-]+:jar:[^:]+:test  # HADOOP-20014: Jetty jars only the Jetty test provider pulled in

# --- HADOOP-20012: catalog-webapp and timelineservice-hbase tests stop
# pulling third-party Jetty and the old servlet API. Test scope only.
dep-removed hadoop-yarn-project/hadoop-yarn/hadoop-yarn-applications/hadoop-yarn-applications-catalog/hadoop-yarn-applications-catalog-webapp\|org\.eclipse\.jetty(\.http2)?:[a-z0-9\-]+:jar:[^:]+:test  # HADOOP-20012
dep-removed hadoop-yarn-project/hadoop-yarn/hadoop-yarn-applications/hadoop-yarn-applications-catalog/hadoop-yarn-applications-catalog-webapp\|javax\.servlet:javax\.servlet-api:jar:[^:]+:test  # HADOOP-20012
# The javax.servlet-api exclusions added to timelineservice-hbase-tests show
# no difference in CI.

# --- HADOOP-20014: hadoop-client-minicluster shades the JDK HTTP test
# container instead of the Jetty one. Two of its changes show no difference
# in CI: jetty-util, now declared directly, resolves as before, and the new
# com/sun/net/httpserver shade exclude leaves the jar's entries unchanged.
dep-(removed|added) hadoop-client-modules/hadoop-client-minicluster\|org\.glassfish\.jersey\.(test-framework\.providers|containers):jersey-(test-framework-provider|container)-(jetty|jetty-http|jdk-http):jar:[^:]+:compile  # HADOOP-20014
jar-entry-(removed|added|changed) share/hadoop/client/hadoop-client-minicluster-[^/]+\.jar!org/apache/hadoop/shaded/org/glassfish/jersey/(test/(jetty|jdkhttp)|jetty|jdkhttp)/.*  # HADOOP-20014: shaded test container classes
jar-entry-(removed|added|changed) share/hadoop/client/hadoop-client-minicluster-[^/]+\.jar!META-INF/(services/.*|versions/.*/org/apache/hadoop/shaded/org/glassfish/jersey/.*|maven/org\.glassfish\.jersey\..*)  # HADOOP-20014: service files and metadata of the swapped container
class-(removed|added) client-minicluster:org\.apache\.hadoop\.shaded\.org\.glassfish\.jersey\.(test\.(jetty|jdkhttp)|jetty|jdkhttp)\..*  # HADOOP-20014
# jersey-container-jetty-http depended on jetty-continuation; nothing else in the
# minicluster jar uses it, so its shaded copy goes with the Jetty test container.
dep-removed hadoop-client-modules/hadoop-client-minicluster\|org\.eclipse\.jetty:jetty-continuation:jar:[^:]+:compile  # HADOOP-20014: came with jersey-container-jetty-http
jar-entry-removed share/hadoop/client/hadoop-client-minicluster-[^/]+\.jar!org/apache/hadoop/shaded/org/eclipse/jetty/continuation/.*  # HADOOP-20014: came with jersey-container-jetty-http
