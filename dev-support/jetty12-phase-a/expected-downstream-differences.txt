# Differences in what downstream Maven projects resolve (downstream-deps.sh)
# that phase A is meant to cause. lib/compare_downstream.py reports every
# other difference as UNEXPECTED, and lists the rules below that matched
# nothing.
#
# Format: <kind> <regex>  # reason   (the regex must match the whole subject)
#   downstream-dep-removed / -added   <scenario>/<hadoop artifact>|<group:artifact:type:version:scope>
#   downstream-dep-changed            <scenario>/<hadoop artifact>|<group:artifact:type>:<old version:scope> -> <new version:scope>

# HADOOP-20013: the duplicate javax.annotation-api goes; jakarta.annotation-api,
# which provides the same javax.annotation classes, was already resolved.
downstream-dep-removed [^|]+\|javax\.annotation:javax\.annotation-api:jar:1\.3\.2:(compile|runtime|test)  # HADOOP-20013: duplicate javax.annotation jar dropped

# HADOOP-20016: hadoop-common and the shuffle handler declare jetty-http
# directly, so a project that depends on them gets it one level nearer than
# through jetty-server. Next to another jetty-http at that level (the older
# Jetty client of the older-jetty scenario), Maven now takes the one declared
# first, Hadoop's, instead of the nearer one. Either way that project mixes
# Jetty versions (jetty-io still comes from the client); only which jar wins
# changes.
downstream-dep-changed older-jetty/(hadoop-common|hadoop-mapreduce-client-shuffle)\|org\.eclipse\.jetty:jetty-http:jar:[^:]+:compile -> [^:]+:compile  # HADOOP-20016: jetty-http nearer to downstream projects
